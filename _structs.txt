struct Entidad 
    int id 
    Vector2 posicion 
    Vector3 rotacion 
    bool visible 
    void Mover ( float dx , float dy ) 
        posicion . x = posicion . x + dx 
        posicion . y = posicion . y + dy 
    bool EstaVisible ( ) 
        return visible 
struct Personaje : Entidad 
    int nivel 
    bool vivo 
    String8 nombre 
    String8 apellido 
    float edad 
    String64 descripcion 
    int8 estado 
    Vector2 velocidad 
    void Start ( ) 
        vivo = true 
        estado = int8 . activo 
        nivel = Config . nivel_basico 
        velocidad . x = 0 . 0 
        velocidad . y = 0 . 0 
    void Update ( float dt ) 
        if vivo == true 
            nivel = Mapa . id_del_nivel + 1 
            velocidad . y = velocidad . y - Config . gravedad * dt 
            Mover ( velocidad . x * dt , velocidad . y * dt ) 
            if nivel > Config . nivel_maximo 
                nivel = Config . nivel_maximo 
        else 
            estado = int8 . inactivo 
    void Damage ( int cantidad ) 
        if cantidad <= 0 
            return 
        edad = edad - cantidad 
        if edad <= 0 . 0 
            Destroy ( ) 
        else 
            estado = int8 . pausado 
    void Destroy ( ) 
        vivo = false 
        estado = int8 . destruido 
        visible = false 
    String8 ObtenerNombreCompleto ( ) 
        String8 resultado 
        resultado = nombre +   + apellido 
        return resultado 
    bool EsAdulto ( ) 
        return edad >= 18 . 0 
struct Inventario 
    int capacidad 
    int cantidad_actual 
    Personaje duenio 
    bool abierto 
    void Agregar ( int item_id ) 
        if cantidad_actual < capacidad 
            cantidad_actual = cantidad_actual + 1 
    void Quitar ( int item_id ) 
        if cantidad_actual > 0 
            cantidad_actual = cantidad_actual - 1 
    bool EstaLleno ( ) 
        return cantidad_actual >= capacidad 
    void Vaciar ( ) 
        cantidad_actual = 0 
struct Juego 
    Config config 
    Mapa mapa 
    Personaje jugador 
    Inventario inventario 
    void Inicializar ( ) 
        config . nivel_basico = 1 
        config . nivel_maximo = 99 
        config . gravedad = 9 . 8 
        config . nombre_juego = Mi Juego 
        config . debug = false 
        mapa . id_del_nivel = 1 
        mapa . nombre = Nivel Inicial 
        mapa . dificultad = 1 . 5 
        jugador . nombre = Juan 
        jugador . apellido = Perez 
        jugador . edad = 25 . 0 
        jugador . Start ( ) 
        inventario . capacidad = 20 
        inventario . cantidad_actual = 0 
        inventario . duenio = jugador 
        inventario . abierto = false 
    void Tick ( float dt ) 
        jugador . Update ( dt ) 
        if jugador . vivo == false 
            Reiniciar ( ) 
    void Reiniciar ( ) 
        jugador = Personaje 
        jugador . Start ( ) 
        inventario . Vaciar ( ) 
    void Loop ( ) 
        float dt = 0 . 016 
        int contador = 0 
        while jugador . vivo == true 
            Tick ( dt ) 
            contador = contador + 1 
            if contador > 1000 
                break 
        for int i = 0 ; i < 10 ; i = i + 1 
            jugador . Damage ( 1 ) 
    int CalcularPuntaje ( ) 
        int puntaje = 0 
        puntaje = jugador . nivel * 100 
        puntaje = puntaje + inventario . cantidad_actual * 10 
        if jugador . EsAdulto ( ) == true 
            puntaje = puntaje + 50 
        return puntaje
